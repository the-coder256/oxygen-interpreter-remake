STMT = | IDENT "(" (EXPR ("," EXPR)*)? ")"
       | NAME "=" EXPR
       | "if" EXPR "{" STMT* "}" ("else" "if" EXPR "{" STMT* "}")* ("else" "{" STMT* "}")?
       | "define" NAME "(" (NAME ("," NAME)*)? ")" "{" STMT* "}"
       | "return" EXPR

EXPR = | NUMBER
       | STRING
       | IDENT
       | IDENT "(" (EXPR ("," EXPR)*)? ")"
       | COMP

COMP = | MATH (("<" | ">" | "==" | "<=" | ">=") MATH)*

MATH = | TERM (("+" | "-") TERM)*

TERM = | FACTOR (("*" | "/") FACTOR)*

FACTOR = | (EXPR | "(" EXPR ")")
