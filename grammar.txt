STMT = | IDENT "(" (EXPR ("," EXPR)*)? ")"
       | NAME "=" EXPR
       | "if" EXPR "{" STMT* "}" ("else" "{" STMT* "}")?
       | "define" NAME "(" (NAME ("," NAME)*)? ")" "{" STMT* "}"
       | "return" EXPR

EXPR = | NUMBER
       | STRING
       | IDENT
       | IDENT "(" (EXPR ("," EXPR)*)? ")"
